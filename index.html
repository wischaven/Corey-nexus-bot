// Called on every completed candle
function onCandle(candle, indicators, portfolio) {
  // candle: { t, o, h, l, c, v }
  // indicators: {
  //   rsi, bb:{upper,mid,lower,pct},
  //   macd:{macd,signal,hist},
  //   atr, ema9, ema21, vwap,
  //   stochRsi:{k,d}, confluence
  // }
  // portfolio: {
  //   cash, position, entryPrice,
  //   equity, trades
  // }
  // Return: "buy" | "sell" | null

  if (indicators.rsi < 30 &&
      indicators.bb.pct < 0.2 &&
      !portfolio.position) {
    return "buy";
  }
  if (indicators.rsi > 70 &&
      portfolio.position) {
    return "sell";
  }
  return null;
}
